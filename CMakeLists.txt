CMAKE_MINIMUM_REQUIRED(VERSION 3.24)

# === Configuration and Utilities ===
INCLUDE(${CMAKE_CURRENT_SOURCE_DIR}/Configuration.cmake)

# === Project Definition ===
PROJECT(
  Fleur
  VERSION 0.1
  DESCRIPTION "Fleur Engine Sandbox Project"
  LANGUAGES CXX
)

# === C++ Standard Configuration ===
SET(CMAKE_CXX_STANDARD ${FL_CPP_LANG_VER})
SET(CMAKE_CXX_STANDARD_REQUIRED ON)
SET(CMAKE_CXX_EXTENSIONS OFF)

# === Build Configuration ===
SET(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Parallel build configuration
IF(NOT DEFINED CMAKE_BUILD_PARALLEL_LEVEL)
  # Default to number of processors if not specified
  INCLUDE(ProcessorCount)
  PROCESSORCOUNT(N)
  IF(NOT N EQUAL 0)
    SET(CMAKE_BUILD_PARALLEL_LEVEL ${N})
  ELSE()
    # Fallback to 8 if processor count fails
    SET(CMAKE_BUILD_PARALLEL_LEVEL 8)
  ENDIF()
ENDIF()

MESSAGE(STATUS "Build parallel level: ${CMAKE_BUILD_PARALLEL_LEVEL}")

# === MSVC Runtime Configuration ===
IF(MSVC)
  IF(FL_CONF_DEBUG)
    SET(CMAKE_MSVC_RUNTIME_LIBRARY ${FL_MSVC_RUNTIME_DEBUG})
  ELSEIF(FL_CONF_RELEASE)
    SET(CMAKE_MSVC_RUNTIME_LIBRARY ${FL_MSVC_RUNTIME_RELEASE})
  ENDIF()
ENDIF()

# === Output Directories ===
foreach(OUTPUTCONFIG ${CMAKE_CONFIGURATION_TYPES})
    string(TOUPPER ${OUTPUTCONFIG} OUTPUTCONFIG_UPPER)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${OUTPUTCONFIG_UPPER} "${CMAKE_BINARY_DIR}/${OUTPUTCONFIG}/bin" CACHE PATH "" FORCE)
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${OUTPUTCONFIG_UPPER} "${CMAKE_BINARY_DIR}/${OUTPUTCONFIG}/lib" CACHE PATH "" FORCE)
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${OUTPUTCONFIG_UPPER} "${CMAKE_BINARY_DIR}/${OUTPUTCONFIG}/lib" CACHE PATH "" FORCE)
    set(CMAKE_PDB_OUTPUT_DIRECTORY_${OUTPUTCONFIG_UPPER}     "${CMAKE_BINARY_DIR}/${OUTPUTCONFIG}/bin" CACHE PATH "" FORCE)
endforeach()

# === IDE Configuration ===
IF(CMAKE_PROJECT_NAME STREQUAL PROJECT_NAME)
  SET_PROPERTY(GLOBAL PROPERTY USE_FOLDERS ON)
ENDIF()

# === Global Variables ===
SET(ORGANIZATION "FLEUR")

# === Platform Definitions ===
IF(FLEUR_PLATFORM STREQUAL "macos")
  ADD_COMPILE_DEFINITIONS(FLEUR_PLATFORM_MACOS)
ELSEIF(FLEUR_PLATFORM STREQUAL "x64")
  ADD_COMPILE_DEFINITIONS(FLEUR_PLATFORM_WIN)
ENDIF()

# === Testing Configuration ===
ENABLE_TESTING()

# === Subdirectories ===
ADD_SUBDIRECTORY(MemoryManager)
ADD_SUBDIRECTORY(CoreLib)
ADD_SUBDIRECTORY(Engine)
ADD_SUBDIRECTORY(Sandbox)

IF(ENABLE_FLEUR_TEST)
  ADD_SUBDIRECTORY(Tests)
ENDIF()
